Setup Filepond with React JS


        npm install filepond react-filepond filepond-plugin-image-preview filepond-plugin-file-validate-type --save
    

        import React, { useState } from 'react';
        import { FilePond, registerPlugin } from 'react-filepond';
        import 'filepond/dist/filepond.min.css';
        import FilePondPluginImagePreview from 'filepond-plugin-image-preview';
        import 'filepond-plugin-image-preview/dist/filepond-plugin-image-preview.css';

        // Register the plugins (e.g., image preview)
        registerPlugin(FilePondPluginImagePreview);

        const FileUploadComponent = () => {
            const [files, setFiles] = useState([]);

            return (
                
                div className="App">
                    FilePond
                        files={files}
                        onupdatefiles={setFiles}
                        allowMultiple={true}
                        server="http://your-laravel-app-url/api/upload" // Configure the Laravel endpoint
                        name="files" /* sets the file input name to 'files' */
                        labelIdle='Drag & Drop your files or [Browse] '
                    />
                /div>
            );
        };

        export default FileUploadComponent;

Laravel Backend Setup

Define routes: In your Laravel application, set up a route (usually in routes/api.php) to handle the POST requests from FilePond.


        use Illuminate\Support\Facades\Route;
        use App\Http\Controllers\UploadController;

        Route::post('/upload', [UploadController::class, 'process'])->name('upload.process');
        Route::delete('/upload', [UploadController::class, 'revert'])->name('upload.revert');

Implement upload logic: In the UploadController, handle the file saving process. FilePond sends the file as multipart/form-data in a POST request by default.


        namespace App\Http\Controllers;

        use Illuminate\Http\Request;
        use Illuminate\Support\Facades\Storage;

        class UploadController extends Controller
        {
            public function process(Request $request)
            {
                if ($request->hasFile('files')) { // The input name matches the 'name' prop in React
                    $file = $request->file('files');
                    // Store the file temporarily and get its path/identifier
                    $path = $file->store('public/tmp');
                    // Return the temporary path to FilePond
                    return response($path, 200)->header('Content-Type', 'text/plain');
                }
                return response('No file uploaded', 400);
            }

            public function revert(Request $request)
            {
                // Handle file removal from the temp directory if the user cancels
                $filePath = $request->getContent();
                Storage::delete($filePath);
                return response(null, 200);
            }
        }
